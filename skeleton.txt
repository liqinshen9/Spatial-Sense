How to create a React + TypeScript + .NET 10.0 skeleton app?

1. mkdir project-name
   cd project-name


2. dotnet new webapi --use-controllers --framework net10.0 --no-https -o backend -n Backend
   cd backend


3. For a minimal backend, use the following Program.cs:

var builder = WebApplication.CreateBuilder(args);

builder.WebHost.UseUrls("http://localhost:5000");

builder.Services.AddCors(options =>
{
    options.AddPolicy("Frontend", policy =>
    {
        policy
            .WithOrigins("http://localhost:5173")
            .AllowAnyHeader()
            .AllowAnyMethod();
    });
});

var app = builder.Build();

app.UseCors("Frontend");

app.MapGet("/", () => "Backend is running!");

app.MapGet("/api/message", () =>
{
    return new
    {
        message = "Hello from the .NET backend!"
    };
});

app.Run();

4. Run backend: dotnet run
Backend runs at: http://localhost:5000

5. cd ..
npm create vite@latest frontend -- --template react-ts
cd frontend
npm install

6. Then in frontend/src/App.tsx, call the backend:

import { useEffect, useState } from "react";

function App() {
  const [message, setMessage] = useState("Loading...");

  useEffect(() => {
    fetch("http://localhost:5000/api/message")
      .then((response) => response.json())
      .then((data) => setMessage(data.message))
      .catch(() => setMessage("Could not connect to backend."));
  }, []);

  return (
    <main>
      <h1>React + .NET Test</h1>
      <p>{message}</p>
    </main>
  );
}

export default App;

7. npm run dev
Frontend runs at: http://localhost:5173



Summary: 
backend  → .NET API       → dotnet run      → localhost:5000
frontend → React/Vite app → npm run dev     → localhost:5173
